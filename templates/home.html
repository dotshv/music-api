Four endpoints. That's the whole API.

Every endpoint is a GET request. Responses are JSON, except the stream which returns audio.

GET/songs

Search the catalog by song title or artist.

 GET/songs/{id}

Get title, artist and cover art for one song.

 GET/songs/{id}/stream

Stream the audio. Use it as the source for a player or a voice bot.

 GET/resolve

Turn a song title or a Spotify link into a single song.

Built for bots and apps.

# Save the audio
curl -L "{{ base_url }}/api/v1/songs/{id}/stream" -o song.audio
import requests

BASE = "{{ base_url }}/api/v1"

r = requests.get(f"{BASE}/songs", params={"query": "kesariya", "limit": 1}, timeout=10)
r.raise_for_status()

song = r.json()["results"][0]
print(song["title"], "-", song["artist"])
print(song["stream_url"])
const BASE = "{{ base_url }}/api/v1";

const res = await fetch(`${BASE}/songs?query=kesariya&limit=1`);
if (!res.ok) throw new Error((await res.json()).message);

const { results } = await res.json();
const audio = new Audio(results[0].stream_url);
audio.play();
